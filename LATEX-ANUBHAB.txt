\subsection*{Method of Seperation of variables to solve PDE :}
\textbf{Laplaces Equation :}
\be {\nabla^2} V = 0 \ee
\be \vec{\nabla}.\vec{E} = {1\over{\epsilon_{0}}}\rho ~~~ and ~~~ \vec{E} = \nabla V\ee
\bc where $V(x,y,z)$ is eclectrostatic potential \ec

\textbf{Poisson's Equation :}
\be \nabla^{2}V(\vec{r}) = -{{\rho(\vec{r})}\over{\epsilon_{0}}} \ee
\bc where $\nabla^{2}$ is Laplacian \ec
\be \nabla^{2}V = 0 \ee
\begin{equation} \implies {{\partial^{2}V}\over{\partial x^{2}}} +
{{\partial^{2}V}\over{\partial y^{2}}} + {{\partial^{2}V}\over{\partial z^{2}}} = 0
\end{equation}
\bc This is the Elliptic Partial differential equation \ec
\textbf{Wave Equation :}\\
for $\mid{\psi(x,t)}\mid$ not large :
\be
{{\partial^{2}\psi}\over{\partial x^{2}}}
~=~{1\over{c^2}}{{{\partial^{2}\psi}\over{\partial t^{2}}}} \ee
\be \implies {{{\partial^{2}\psi}\over{\partial x^{2}}} {1\over{c^2}}{{{\partial^{2}\psi}\over{\partial t^{2}}}}}~=~{0}
\ee
c is the velocity of the wave $\sqrt{T\over{m}}$ \\
A wave in 3 dimensions
\be {\nabla^{2}\psi}(\vec{r},t) ~=
~{1\over{c^{2}}}{{\partial^{2}\psi(\vec{r},t)}\over{\partial t^{2}}} \ee
\begin{equation} \implies {{\partial^{2}\psi}\over{\partial x^{2}}} +
{{\partial^{2}\psi}\over{\partial y^{2}}} + {{\partial^{2}\psi}\over{\partial
z^{2}}} - {1\over{c^2}}{{{\partial^{2}\psi}\over{\partial t^{2}}}}~ =~ 0
\end{equation}
\bc This is the Hyperbolic Partial differential equation \ec
